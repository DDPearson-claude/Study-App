### The pipeline
Controls register themselves; their values form `params` = `{id: value}`.
```
K.model(fn)        fn(params) -> outputs {name: number | string}. Pure: no DOM, no lists or objects.
K.update(fn)       fn(params, outputs): draw everything here. More than one is allowed.
K.at(over)         -> outputs of the model at those params (others take each control's starting value)
K.check(label, fn, {source?})   fn() must return exactly true
K.ready()          call once, last: draws the opening state and runs the checks
K.complete()       at the lesson's end
K.params()  K.outputs()  K.refresh()   (refresh reruns the pipeline after you change your own state)
```
A readout whose id is an output key updates itself.

  // ---------- text with amber key terms and italic letters (built with K.el, never innerHTML) ----------
  function letters(el, s) {
    s.split(/(\b[xR]\b)/).forEach((b) => { if (!b) return; el.appendChild(/^[xR]$/.test(b) ? K.el('span', { class: 'v' }, b) : document.createTextNode(b)); });

  function pieces(el, text) {
    const re = /\*([^*]+)\*/g;
    let at = 0, m;
    while ((m = re.exec(text))) {
      if (m.index > at) letters(el, text.slice(at, m.index));
      el.appendChild(K.el('span', { class: 'term' }, m[1]));
      at = m.index + m[0].length;
    }
    if (at < text.length) letters(el, text.slice(at));
  }
  // A sum stays together on one line ("7x − 2x + 3 − x", "3 × 4 = 12", "10 Ω"), with the stop after
  // it; a long one may break only before an equals sign. Its letters are italic.
  const TOK = '(?:−?\\d+(?:\\.\\d+)?(?:[xR][²³]?)?(?: Ω)?|−?\\b[xR]\\b[²³]?|\\?)';
  const GLUE = new RegExp('(?:[×÷−+] )?' + TOK + '(?: [+−=×÷] ' + TOK + ')*[.,?!;]?', 'g');
  function mathBit(el, b) {
    const box = / /.test(b) ? el.appendChild(K.el('span', { class: 'nw' })) : el;
    b.split(/([xR])/).forEach((p) => { if (!p) return; box.appendChild(/^[xR]$/.test(p) ? K.el('span', { class: 'v' }, p) : document.createTextNode(p)); });
  }
  function write(el, text) {
    el.textContent = '';
    let at = 0, m;
    GLUE.lastIndex = 0;
    while ((m = GLUE.exec(text))) {
      if (!m[0].length) { GLUE.lastIndex += 1; continue; }
      if (/^\?[.,?!;]?$/.test(m[0])) continue;   // a question mark ending a sentence
      if (m.index > at) pieces(el, text.slice(at, m.index));
      const bits = m[0].length > 18 ? m[0].split(/ (?==)/) : [m[0]];
      bits.forEach((b, k) => { if (k) el.appendChild(document.createTextNode(' ')); mathBit(el, b); });
      at = m.index + m[0].length;
    }
    if (at < text.length) pieces(el, text.slice(at));
  }

  // ---------- drawing ----------
  const svg = document.getElementById('scene'), art = document.getElementById('art'), ui = document.getElementById('ui');
  let into = null;                                   // where shapes go: the picture, or a part that is arriving

  // How far a part due at t0 has arrived (K.arrive), noting the timetable for the timing check.
  let latest = 0, earliest = Infinity;
  function arr(t0, d) {
    d = d == null ? K.SOFT : d;
    if (t0 + d > latest) latest = t0 + d;
    if (t0 < earliest) earliest = t0;
    return K.arrive(T, t0, d);

  // A part arriving: drawn into a group faded in to a and moved `rise` units into place; nothing before it starts.
  function part(a, rise, draw) {
    if (a <= 0) return;
    const g = add(K.svg('g'));
    K.fade(g, a, rise);
    const keep = into; into = g; draw(); into = keep;
  }
  // A part leaving: faded out as a goes 0 -> 1.
  function leave(a, draw) {

  const r3 = (v) => Math.round(v * 1000) / 1000;
  const rect = (cls, x, y, w, h, rx, attrs) => add(K.svg('rect', Object.assign({ class: cls, x: r3(x), y: r3(y), width: r3(Math.max(0, w)), height: r3(Math.max(0, h)), rx: r3(rx || 0) }, attrs || {})));
  const line = (cls, x1, y1, x2, y2) => add(K.svg('line', { class: cls, x1: r3(x1), y1: r3(y1), x2: r3(x2), y2: r3(y2) }));
  // Text with every letter x or R in italics.
  function text(x, y, s, cls, anchor) {
    const t = K.svg('text', { class: cls, x: r3(x), y: r3(y), 'text-anchor': anchor || 'middle' });
    String(s).split(/([xR])/).forEach((p) => { if (p) t.appendChild(/^[xR]$/.test(p) ? K.svg('tspan', { class: 'v' }, p) : document.createTextNode(p)); });
    return add(t);
  }
  // Spaces that SVG keeps (and measures).
  const nb = (s) => String(s).replace(/ /g, ' ');
  const widths = {};
  function tw(s, cls) {
    const key = cls + '|' + s;
    if (widths[key]) return widths[key];
    const t = svg.appendChild(K.svg('text', { class: cls, x: 0, y: -50 }, String(s)));
    let w = 0;
    try { w = t.getComputedTextLength(); } catch (e) { w = 0; }
    svg.removeChild(t);
    if (w > 0) { widths[key] = w; return w; }
    return String(s).length * 15 * 0.6;
  }
  // Labels in a line, each as near the middle of its span as it can sit without touching the next.
  function pack(items, lo, hi) {
    const order = items.map((it, k) => k).sort((a, b) => items[a].c - items[b].c), left = [];
    let prev = -Infinity;
    order.forEach((k) => { const it = items[k]; left[k] = Math.max(it.c - it.w / 2, lo, prev + 8); prev = left[k] + it.w; });
    let lim = hi;
    order.slice().reverse().forEach((k) => { const it = items[k]; if (left[k] + it.w > lim) left[k] = lim - it.w; lim = left[k] - 8; });
    return left;
  }
  // Term names over a row: a bracket over each term's pieces, its name above.
  function labelSet(items, ly, by, bad) {
    const lefts = pack(items.map((it) => ({ c: (it.a0 + it.a1) / 2, w: tw(nb(it.text), 'tl') })), 4, 336);
    return items.map((it, m) => () => {
      if (it.a1 - it.a0 >= 8) add(K.svg('path', { class: 'brk' + (bad ? ' bad' : ''), d: 'M' + r3(it.a0 + 1.5) + ' ' + (by + 5) + ' V' + by + ' H' + r3(it.a1 - 1.5) + ' V' + (by + 5) }));
      text(lefts[m], ly, nb(it.text), 'tl' + (bad ? ' bad' : ''), 'start');
    });
  }
  // One piece: a bar with its letter (or, once x is tried, its value) inside, or a square.
  function drawPiece(p, x, bt, w, U, bad, attrs, tl, xr) {
    if (p.kind === 'sq') { const s = Math.max(2, U - 2); rect('sq' + bad, x + (U - s) / 2, bt + BH - s, s, s, Math.min(2, s / 4), attrs); return; }
    rect('xb' + bad, x + 0.75, bt, w - 1.5, BH, Math.min(3, w / 4), attrs);
    if (w < 15) return;
    const cx = x + w / 2, cy = bt + BH / 2 + 5;
    if (!tl || tl.sub == null) { text(cx, cy, p.v, 'il'); return; }
    leave(arr(tl.sub, 0.25), () => text(cx, cy, p.v, 'il'));
    part(arr(tl.sub + 0.25), 3, () => text(cx, cy, String(xr), 'il'));
  }

  // When piece j of row k arrives (1: there from the start).
  function pieceA(d, tl, k, j) {
    const g = tl.groups[k];
    if (!g) return 1;
    let n;
    if (d.split && d.split[k]) { let acc = 0; n = d.split[k].findIndex((c) => (acc += c) > j); }
    else n = geo(d).info[k].groups.findIndex((gr) => gr.pieces.indexOf(j) >= 0);
    return arr(g[n]);
  }
  // x now: Dan's setting on a slider screen, else where the demonstration has slid it, else the step's.
  function xNow(d, st) {
    if (!d.slide) return d.x;
    if (st.x != null) return st.x;
    return K.lerp(d.slide.from, d.slide.to, arr(d.slide.t, d.slide.d));
  }
  // A row's value shows once due, and hides while x slides.
  function valVis(tl, k) {
    if (tl.val[k] == null) return 0;
    let a = arr(tl.val[k]);
    if (tl.slide) { const s = tl.slide; a *= 1 - arr(s.t, 0.25) * (1 - arr(s.t + s.d, 0.4)); }
    return a;
  }
  // Bars and squares in words.
  function pieceWords(ts, letter) {
    const b = ts.filter((t) => t.v).reduce((a, t) => a + t.c, 0), s = ts.filter((t) => !t.v).reduce((a, t) => a + t.c, 0), out = [];
    if (b) out.push(b === 1 ? 'one bar ' + letter + ' long' : b + ' bars, each ' + letter + ' long');
    if (s) out.push(s === 1 ? 'one square' : s + ' squares');
    return out.join(' and ') || 'nothing';
  }

  // One row of the question: its pieces, what is taken away, the term names, its value and any
  // strip of squares under it.
  function drawRow(d, st, tl, G, k, xv) {
    const info = G.info[k], U = G.U, y0 = G.rowY[k], bt = y0 + 29, P = positions(info, xv, U), xr = Math.round(xv);
    const tT = tl.tidy[k], letter = d.letter || 'x';
    const cutA = tT == null ? 0 : arr(tT, K.SOFT), slideA = tT == null ? 0 : arr(tT + 0.6, 0.6);
    info.pieces.forEach((p, j) => {
      const q = P.list[j], x = K.lerp(q.x0, q.x1, slideA), a = pieceA(d, tl, k, j);
      const draw = () => drawPiece(p, x, bt, q.w, U, '', { 'data-row': k }, tl, xr);
      if (p.cut != null && tT != null) leave(cutA, () => part(a, 6, draw));
      else part(a, 6, draw);
    });
    // what is taken away: a dashed outline under the pieces it takes, named under it
    const ny = bt + BH + 5;
    const negs = info.negs.map((n, m) => {
      const a = n.covers.map((j) => P.list[j]), last = a[a.length - 1];
      return { a0: a[0].x0, a1: last.x0 + last.w, a: tl.negs[k] ? arr(tl.negs[k][m]) : 1, text: '− ' + termText(info.terms[n.term]) };
    });
    const nl = pack(negs.map((it) => ({ c: (it.a0 + it.a1) / 2, w: tw(nb(it.text), 'tl') })), 4, 336);
    negs.forEach((it, m) => {
      const shown = () => part(it.a, 6, () => { rect('neg', it.a0 + 1, ny, it.a1 - it.a0 - 2, 16, 3, { 'data-row': k }); text(nl[m], ny + 32, nb(it.text), 'tl', 'start'); });
      if (tT != null) leave(cutA, shown); else shown();
    });
    // the term names: as written, then (after a tidy-up) the tidied terms
    const ly = y0 + 15, by = y0 + 22;
    const old = info.groups.map((g, n) => {
      const f = P.list[g.pieces[0]], l = P.list[g.pieces[g.pieces.length - 1]];
      return { a0: K.lerp(f.x0, f.x1, slideA), a1: K.lerp(l.x0, l.x1, slideA) + l.w, text: (n === 0 ? '' : '+ ') + termText(info.terms[g.term]), a: pieceA(d, tl, k, g.pieces[g.pieces.length - 1]) };
    });
    const oldDraw = labelSet(old, ly, by, false);
    old.forEach((it, m) => { const sh = () => part(it.a, 6, oldDraw[m]); if (tT != null) leave(arr(tT, 0.3), sh); else sh(); });
    let tidyText = null;
    if (tT != null) {
      const ts = tidy(info.terms), runs = [];
      info.pieces.forEach((p, j) => { if (p.cut != null) return; const r = runs[runs.length - 1]; if (r && info.pieces[r[0]].kind === p.kind) r.push(j); else runs.push([j]); });
      const neu = runs.map((r, n) => { const l = P.list[r[r.length - 1]]; return { a0: P.list[r[0]].x1, a1: l.x1 + l.w, text: (n === 0 ? '' : '+ ') + termText(ts[n]) }; });
      const na = arr(tT + 1.2);
      labelSet(neu, ly, by, false).forEach((f) => part(na, 6, f));
      tidyText = fmt(ts);
    }
    // its value at its end (or a question mark until a puzzle is answered)
    const end = X0 + K.lerp(P.solid, P.net, slideA), vy = bt + 20;
    if (tl.q[k]) leave(tl.val[k] == null ? 0 : arr(tl.val[k] - 0.25, 0.25), () => text(end + 7, vy, '?', 'val', 'start'));
    const vv = valVis(tl, k), value = valueOf(info.terms, xr);
    if (vv > 0) part(vv, 4, () => text(end + 7, vy, String(value), 'val', 'start'));
    // a strip of squares under the row, each worth 1 (in the warning colour for a wrong answer)
    const sz = Math.max(2, U - 2), sy = ny;
    tl.strips.filter((s) => s.row === k).forEach((s) => {
      let from = 0;
      s.n.forEach((n, m) => {
        const lo = from;
        part(arr(s.at[m]), 6, () => { for (let j = lo; j < n; j++) rect('sq' + (s.bad ? ' bad' : ''), X0 + j * U + (U - sz) / 2, sy, sz, sz, Math.min(2, sz / 4), { 'data-strip': k }); });
        from = n;
      });
      const n = s.n[s.n.length - 1];
      if (s.label) part(arr(s.at[s.at.length - 1]), 6, () => text(X0 + n * U + 7, sy + sz / 2 + 6, String(n), 'val' + (s.bad ? ' bad' : ''), 'start'));
      REC.strips.push({ row: k, end: X0 + n * U, n, bad: !!s.bad, seen: arr(s.at[s.at.length - 1]) >= 0.5 });
    });
    const tidiedNow = !info.negs.length || slideA >= 1;
    REC.rows.push({ end, value: valueOf(info.terms, xv), solid: solidUnits(info, xv), tidied: tidiedNow });
    // in words
    const shownAll = info.pieces.every((p, j) => pieceA(d, tl, k, j) >= 0.5);
    const ts = slideA >= 0.5 ? tidy(info.terms) : info.terms.filter((t) => t.c > 0);
    const negSaid = negs.filter((it) => it.a >= 0.5 && cutA < 0.5).map((it) => say(it.text));
    return (shownAll ? 'A row for ' + say(slideA >= 0.5 && tidyText ? tidyText : info.e) + ', ' + pieceWords(ts, letter) +
      (negSaid.length ? ', with ' + negSaid.join(' and ') + ' marked as taken away' : '') +
      (vv >= 0.5 ? ', making ' + value : tl.q[k] ? ', its value a question mark' : '') + '.' : '');
  }
  // The answer's own row under the question: lining up when right, in the warning colour when not.
  function drawAns(d, tl, G) {
    const r = rowInfo(tl.ans.e), U = G.U, y0 = G.ansY, bt = y0 + 29, P = positions(r, d.x, U), bad = tl.ans.bad ? ' bad' : '';
    const a = arr(tl.ans.t);
    part(a, 6, () => {
      r.pieces.forEach((p, j) => drawPiece(p, P.list[j].x0, bt, P.list[j].w, U, bad, { 'data-row': 'ans' }, null, d.x));
      const items = r.groups.map((g, n) => { const f = P.list[g.pieces[0]], l = P.list[g.pieces[g.pieces.length - 1]]; return { a0: f.x0, a1: l.x0 + l.w, text: (n === 0 ? '' : '+ ') + termText(r.terms[g.term]) }; });

    });

    return a >= 0.5 ? ' Under it, the answer ' + say(tl.ans.e) + ' as its own row, ' + pieceWords(r.terms, d.letter || 'x') + (tl.ans.bad ? ', which does not line up.' : ', which lines up.') : '';
